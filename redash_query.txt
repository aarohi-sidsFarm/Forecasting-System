WITH target AS (
    SELECT '{{target_date}}'::date AS target_date
),

/* ── Customers on holiday ── */
customer_holidays AS (
    SELECT DISTINCT ch.cust_id
    FROM customer_holiday ch
    CROSS JOIN target t
    WHERE ch.is_deleted = false
      AND t.target_date BETWEEN ch.valid_from AND ch.valid_upto
),

/* ══════════════════════════════════════════
   SUBSCRIPTION ORDERS
   FIXES:
   ✅ zero qty filtering
   ✅ alternate NULL fix
   ✅ strict status filtering
   ✅ custom off-day filtering
   ══════════════════════════════════════════ */
subscription_orders AS (
    SELECT
        s.cust_id AS customer_id,
        s.variant_id,
        s.subscription_type,

        CASE
            /* DAILY */
            WHEN s.subscription_type = 'daily'
                THEN COALESCE((s.data->>'quantity')::int, 0)

            /* ALTERNATE */
            WHEN s.subscription_type = 'alternate'
                THEN CASE
                    WHEN COALESCE(s.valid_from, s.start_date) IS NULL
                        THEN 0

                    WHEN (
                        t.target_date
                        - COALESCE(s.valid_from, s.start_date)::date
                    ) % 2 = 0
                        THEN COALESCE((s.data->>'start_date_qty')::int, 0)

                    ELSE COALESCE((s.data->>'next_day_qty')::int, 0)
                END

            /* CUSTOM */
            WHEN s.subscription_type = 'custom'
                THEN COALESCE(
                    (s.data ->> LOWER(TO_CHAR(t.target_date, 'DY')))::int,
                    (s.data ->> UPPER(TO_CHAR(t.target_date, 'DY')))::int,
                    0
                )

            ELSE 0
        END AS qty

    FROM subscription s
    CROSS JOIN target t

    LEFT JOIN customer_holidays ch
        ON ch.cust_id = s.cust_id

    WHERE s.subscription_type IN ('daily', 'alternate', 'custom')

      /* STRICT STATUS FILTER */
      AND LOWER(TRIM(COALESCE(s.subscription_status, ''))) = 'active'

      AND ch.cust_id IS NULL

      /* Pause logic */
      AND NOT (
            s.pause_start_date IS NOT NULL
            AND s.pause_end_date IS NOT NULL
            AND t.target_date BETWEEN s.pause_start_date
                                  AND s.pause_end_date
          )

      /* Validity */
      AND (
            s.valid_from IS NULL
            OR s.valid_from::date <= t.target_date
          )

      AND (
            s.valid_upto IS NULL
            OR s.valid_upto::date >= t.target_date
          )

      /* REMOVE ZERO-QTY ROWS */
      AND (
            CASE
                WHEN s.subscription_type = 'daily'
                    THEN COALESCE((s.data->>'quantity')::int, 0)

                WHEN s.subscription_type = 'alternate'
                    THEN CASE
                        WHEN COALESCE(s.valid_from, s.start_date) IS NULL
                            THEN 0

                        WHEN (
                            t.target_date
                            - COALESCE(s.valid_from, s.start_date)::date
                        ) % 2 = 0
                            THEN COALESCE((s.data->>'start_date_qty')::int, 0)

                        ELSE COALESCE((s.data->>'next_day_qty')::int, 0)
                    END

                WHEN s.subscription_type = 'custom'
                    THEN COALESCE(
                        (s.data ->> LOWER(TO_CHAR(t.target_date, 'DY')))::int,
                        (s.data ->> UPPER(TO_CHAR(t.target_date, 'DY')))::int,
                        0
                    )

                ELSE 0
            END
          ) > 0
),

/* ══════════════════════════════════════════
   BUY ONCE ORDERS
   FIXES:
   ✅ removed Bulk Order
   ✅ removed fallback leakage
   ══════════════════════════════════════════ */
buyonce_orders AS (
    SELECT
        toi.customer_id,
        toi.variant_id,
        'buy_once' AS subscription_type,
        toi.quantity AS qty

    FROM temp_order_item toi
    CROSS JOIN target t

    WHERE toi.order_date = t.target_date

      AND toi.status IN (
            'DRAFT',
            'PROCESSED_TO_ORDER'
          )

      AND (
            toi.purchase_type IN (
                'Buy Once',
                'Adhoc Buy Once',
                'FIRST_X_ORDER'
            )

            OR toi.purchase_type ILIKE '%buy once%'
          )
),

/* ══════════════════════════════════════════
   COMBINE ALL ORDERS
   ══════════════════════════════════════════ */
all_orders AS (
    SELECT * FROM subscription_orders

    UNION ALL

    SELECT * FROM buyonce_orders
),

/* ══════════════════════════════════════════
   ENRICHMENT
   ══════════════════════════════════════════ */
row_level AS (
    SELECT
        ao.customer_id,
        ao.subscription_type,
        ao.qty,

        pv.price_override_amount AS selling_price,

        ao.qty * pv.price_override_amount AS row_value,

        COALESCE(ca.city_cd, 'UNKNOWN') AS city_cd,

        pp.name AS product_name

    FROM all_orders ao

    JOIN product_productvariant pv
        ON pv.id = ao.variant_id

    JOIN product_product pp
        ON pp.id = pv.product_id

    LEFT JOIN (
        SELECT DISTINCT ON (cust_id)
            cust_id,
            city_cd
        FROM customer_address
        WHERE is_active = true
        ORDER BY cust_id, updated_at DESC
    ) ca
        ON ca.cust_id = ao.customer_id
),

/* ══════════════════════════════════════════
   CUSTOMER TOTAL VALUE
   ══════════════════════════════════════════ */
customer_total AS (
    SELECT
        customer_id,
        SUM(row_value) AS total_order_value
    FROM row_level
    GROUP BY customer_id
),

/* ══════════════════════════════════════════
   WALLET CHECK
   FIX:
   ✅ lowercase wallet_type handling
   ══════════════════════════════════════════ */
wallet_check AS (
    SELECT
        ct.customer_id,

        CASE
            WHEN LOWER(COALESCE(c.wallet_type, '')) = 'prepaid'
             AND ct.total_order_value > c.current_balance
            THEN true

            ELSE false
        END AS low_balance_flag

    FROM customer_total ct

    JOIN customers c
        ON c.id = ct.customer_id
)

/* ══════════════════════════════════════════
   FINAL OUTPUT
   ══════════════════════════════════════════ */
SELECT
    t.target_date,

    r.city_cd,

    r.product_name,

    r.subscription_type,

    /* CONFIRMED QTY */
    SUM(
        CASE
            WHEN w.low_balance_flag = false
            THEN r.qty
            ELSE 0
        END
    ) AS confirmed_qty,

    /* EXPECTED QTY */
    SUM(r.qty) AS expected_qty,

    /* AT RISK QTY */
    SUM(
        CASE
            WHEN w.low_balance_flag = true
            THEN r.qty
            ELSE 0
        END
    ) AS at_risk_qty,

    /* CONFIRMED ORDERS */
    COUNT(
        DISTINCT CASE
            WHEN w.low_balance_flag = false
            THEN r.customer_id
        END
    ) AS confirmed_orders,

    /* TOTAL ORDERS */
    COUNT(DISTINCT r.customer_id) AS total_orders

FROM row_level r

JOIN wallet_check w
    ON w.customer_id = r.customer_id

CROSS JOIN target t

GROUP BY
    t.target_date,
    r.city_cd,
    r.product_name,
    r.subscription_type

ORDER BY
    r.city_cd,
    confirmed_qty DESC;
